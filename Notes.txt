<?php

namespace App\Http\Controllers;

use App\Models\File;
use Error;
use Illuminate\Http\Request;

class FileController extends Controller
{
    public function index(Request $req)
    {
        $rows = File::select()->get();
        return view('adding', compact('rows'));
    }
    public function store(Request $req)
    {
        try {
            $req->validate([
                'file' => 'required|mimes:jpg,png,jpeg|max:2048',
                'name' => 'required|string',
                'price' => 'required',
                //'gallery.*' => 'required|mimes:jpg,jpeg,png|max:2048', // validate each file
            ]);

            $extension = $req->file->extension();

            $file = time() . '_' . rand(1000, 9999) . '.' . $extension;

            // folder auto created
            $req->file->move(public_path('uploads'), $file);

            File::create([
                'name' => $req->name,
                'file' => $file,
                'price' => $req->price,
                'gallery' => $req->price,
            ]);

            //$rows = File::select('id', 'name', 'file', 'price')->get();
            // $rows = File::all();
            // return redirect()->back()->withSuccess('File Upload Success');
            //return view('adding', compact('rows'))->withSuccess('File Upload Success');
            $rows = File::select()->get();
            return redirect()->back()->with([
                'success' => 'File Upload Success',
                'rows' => $rows
            ]);
        } catch (Error $e) {
            return redirect()->back()->withErrors('File Upload Success');
        }
    }
    public function store2(Request $req)
    {
        $req->validate([
            'file' => 'required|mimes:jpg,png,jpeg|max:2048',
            'name' => 'required|string',
            'price' => 'required',
            'gallery.*' => 'required|mimes:jpg,jpeg,png|max:2048', // validate each file
        ]);

        // Field name jisse file aayi hai
        $fieldName = 'file'; // agar multiple fields hain to loop kar sakte ho

        // Unique filename
        $extension = $req->file($fieldName)->extension();
        $fileName = time() . '_' . rand(1000, 9999) . '.' . $extension;

        // Upload path: uploads/{fieldName}
        $uploadPath = public_path('uploads/' . $fieldName);

        // Agar folder exist nahi karta to create karo
        if (!file_exists($uploadPath)) {
            mkdir($uploadPath, 0777, true);
        }

        // File move karo
        $req->file($fieldName)->move($uploadPath, $fileName);

        // Save DB record
        File::create([
            'name' => $req->name,
            'file' => $fileName,
            'price' => $req->price,
            'gallery' => $fieldName, // optional: store folder info
        ]);

        return redirect()->back()->withSuccess('File Upload Success');
    }

    public function delete($id)
    {
        try {
            $file = File::find($id);

            if ($file) {
                $path = public_path('uploads/' . $file->file);

                // Agar actual file exist karti hai to delete karo
                if (file_exists($path)) {        unlink($path);         }

                // DB record delete
                $file->delete();

                // Success message
                return redirect()->back()->with('success', 'File Delete Success');
            } else {
                // Agar record hi nahi mila
                return redirect()->back()->with('error', 'File not found');
            }
        } catch (\Exception $e) {
            // Agar koi exception aayi
            return redirect()->back()->with('error', 'Delete failed: ' . $e->getMessage());
        }
    }


    public function delete2($id)
    {
        $file = File::find($id);
        if ($file) {
            $path = public_path('uploads/' . $file->file);
            if (file_exists($path)) {
                unlink($path);
            }
            $file->delete();
        }
    }
}
