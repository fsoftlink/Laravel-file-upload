# 1. Initialize git in your project folder
git init

# 2. Add your GitHub remote repository (replace with your repo URL)
git remote add origin https://github.com/fsoftlink/React_Js_Practice_Vite.git

# 3. Create a .gitignore file to exclude node_modules and other unwanted files
echo "node_modules/" >> .gitignore
echo ".env" >> .gitignore
echo "dist/" >> .gitignore
echo "notes.txt" >> .gitignore

# 4. Stage all files except ignored ones
git add .


# 5. Commit your changes with a message
git commit -m "Initial commit"

# 6. Rename branch to main (GitHub default)
git branch -M main

# 7. Push to GitHub (first time push)
git push -u origin main


# 1. Set your global Git username (replace with your actual name)
git config --global user.name "Kuldip"

# 2. Set your global Git email (use the same email as your GitHub account)
git config --global user.email "your_email@example.com"

# Example:
git config --global user.name "Kuldip"
git config --global user.email "fsoftlink@gmail.com"
